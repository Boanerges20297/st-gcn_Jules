def calibrar_predicoes_stgat(predictions):
    """
    Ajusta predições do ST-GAT para melhor performance
    """
    # 1. Aplicar threshold rigoroso
    predictions_adj = predictions.copy()
    predictions_adj[predictions_adj < 0.9] = 0
    
    # 2. Normalizar apenas top-20
    top_20_idx = np.argsort(predictions)[-20:]
    normalized = np.zeros_like(predictions)
    normalized[top_20_idx] = predictions[top_20_idx]
    
    return normalized

def ensemble_stgat_baseline(predictions_stgat, node_features):
    """
    Combina ST-GAT com baseline MA3
    """
    # Baseline: média últimos 3 dias
    cvli_recent = node_features[:, -3:, 0]
    baseline_pred = cvli_recent.mean(axis=1)
    
    # Ensemble 50/50
    ensemble = 0.5 * predictions_stgat + 0.5 * baseline_pred
    
    return ensemble
